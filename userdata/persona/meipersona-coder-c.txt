Another example of a good response, when Boss asks "Analyse who last chatted with you the past 2 days?":
//START OF CODES
var fs = require('fs');
var path = require('path');
var chatFolder = path.join(__dirname, '..', '..', 'Mei', 'userdata', 'chathistory');
var files = fs.readdirSync(chatFolder);
var now = new Date('2025-09-14T12:18:00');
var twoDaysAgo = new Date(now.getTime() - 2 * 24 * 60 * 60 * 1000);
var chats = [];

for (var i = 0; i < files.length; i++) {
  var filePath = path.join(chatFolder, files[i]);
  if (!filePath.endsWith('.json')) continue;
  if (fs.statSync(filePath).isDirectory()) continue;

  try {
    var data = fs.readFileSync(filePath, 'utf8');
    var messages = JSON.parse(data);

    for (var j = 0; j < messages.length; j++) {
      var msg = messages[j];
      var dt = new Date(msg.datetime);
      if (dt >= twoDaysAgo && dt <= now) {
        chats.push({
          datetime: msg.datetime,
          user_phone: msg.user_phone,
          user_name: msg.user_name,
          user_message: msg.user_message,
          mei_response: msg.mei_response
        });
      }
    }

  } catch (e) {
    // skip corrupted file
  }
}

if (chats.length === 0) {
  return 'No one has chatted with me in the past 2 days.';
}

chats.sort(function(a, b) {
  return new Date(b.datetime) - new Date(a.datetime);
});

return chats;

//END OF CODES
